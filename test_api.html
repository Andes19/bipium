const form = document.getElementById('myForm'); const statusDiv = document.getElementById('status'); form.addEventListener('submit', async (event) => { event.preventDefault(); statusDiv.innerHTML = 'Отправка...'; // Собираем данные в объект const subject = document.getElementById('subject').value; const fileInput = document.getElementById('file'); const file = fileInput.files[0]; // Подготавливаем данные для отправки let dataToSend = { subject: subject }; // Если есть файл, можно отправить его как base64 (но это для небольших файлов) if (file) { const reader = new FileReader(); reader.onloadend = function() { dataToSend.file = { name: file.name, type: file.type, content: reader.result.split(',')[1] // base64 без заголовка }; sendData(dataToSend); }; reader.readAsDataURL(file); } else { sendData(dataToSend); } async function sendData(data) { try { const response = await fetch('https://bipium.red-red.ru:8441/api/webrequest/site-requests', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(data) }); if (response.ok) { statusDiv.innerHTML = '✅ Успешно!'; form.reset(); } else { statusDiv.innerHTML = '❌ Ошибка: ' + response.status; } } catch (error) { statusDiv.innerHTML = '❌ Ошибка: ' + error.message; } } });
